set(media_fragments)
if(KVM_MEDIA_FFMPEG)
    set(media_fragments "ffmpeg.lf")
endif()
idf_component_register(SRCS "cinepak_tile.c" "live_message.c" "media_decode.c" "live_queue.c"
                       INCLUDE_DIRS "include" PRIV_REQUIRES esp_rom heap LDFRAGMENTS ${media_fragments})

if(NOT CMAKE_BUILD_EARLY_EXPANSION AND KVM_MEDIA_FFMPEG)
    get_filename_component(media_ffmpeg "${KVM_MEDIA_FFMPEG}" ABSOLUTE)
    file(STRINGS "${media_ffmpeg}/config.h" allocator REGEX "^#define MALLOC_PREFIX kvm_av_$")
    file(STRINGS "${media_ffmpeg}/src/libavcodec/h264dec.c" mode_hook REGEX "kvm_media_rr_mode")
    file(STRINGS "${media_ffmpeg}/libavcodec/libavcodec.a" motion_hook
         REGEX "kvm_h264_detach_second_mv_list" LIMIT_COUNT 1)
    if(NOT allocator OR NOT mode_hook OR NOT motion_hook)
        message(FATAL_ERROR "Prepare the ESP32-S3 FFmpeg build with tools/live_view/build_ffmpeg.py")
    endif()
    target_sources(${COMPONENT_LIB} PRIVATE "h264_decode.c" "h264_heap.c")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE KVM_HAVE_FFMPEG)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${media_ffmpeg}" "${media_ffmpeg}/src")
    foreach(lib avformat avcodec avutil)
        target_link_libraries(${COMPONENT_LIB} PRIVATE "${media_ffmpeg}/lib${lib}/lib${lib}.a")
    endforeach()
endif()
if(NOT CMAKE_BUILD_EARLY_EXPANSION AND KVM_MEDIA_PAGED_GUEST)
    if(NOT KVM_MEDIA_FFMPEG)
        message(FATAL_ERROR "Paged card presentation uses KVM_MEDIA_FFMPEG for the original stream geometry probe")
    endif()
    get_filename_component(paged_image "${KVM_MEDIA_PAGED_GUEST}" ABSOLUTE)
    file(READ "${paged_image}" paged_magic LIMIT 4)
    if(NOT paged_magic STREQUAL "KVR1")
        message(FATAL_ERROR "Pass decoder.kvr from tools/live_view/paged_h264/build_guest.py")
    endif()
    set(paged_dir "${CMAKE_CURRENT_LIST_DIR}/../../tools/live_view/paged_h264")
    target_sources(${COMPONENT_LIB} PRIVATE "${paged_dir}/paged_video.c")
    target_include_directories(${COMPONENT_LIB} PUBLIC "${paged_dir}")
    target_compile_definitions(${COMPONENT_LIB} PUBLIC KVM_MEDIA_PAGED_GUEST=1)
    target_add_binary_data(${COMPONENT_LIB} "${paged_image}" BINARY RENAME_TO kvm_paged_decoder)
endif()
