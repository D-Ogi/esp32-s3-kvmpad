set(monitor_sources)
if(CONFIG_KVM_USB_MONITOR_EVAL)
    set(monitor_sources "usb_monitor.c" "monitor_wire.c" "monitor_view.c" "monitor_viewport.c" "usb_jpeg_decode.c")
    if(CONFIG_KVM_USB_GUD OR KVM_GUD_PANEL_CHECK)
        list(APPEND monitor_sources "gud_receiver.c" "gud_panel.c" "gud_view.c")
    endif()
    if(KVM_USB_PICTURE_CHECK)
        list(APPEND monitor_sources "usb_jpeg_check.c" "usb_jpeg_rom_check.c")
    endif()
endif()
idf_component_register(SRCS ${monitor_sources} INCLUDE_DIRS "include"
                       PRIV_REQUIRES live_view ui_theme heap freertos esp_timer)
if(CONFIG_KVM_USB_MONITOR_EVAL)
    idf_component_optional_requires(PRIVATE espressif__esp_new_jpeg)
    if(CONFIG_KVM_USB_GUD OR KVM_GUD_PANEL_CHECK)
        target_compile_definitions(${COMPONENT_LIB} PUBLIC KVM_GUD_PANEL=1)
        if(KVM_GUD_PANEL_CHECK)
            target_compile_definitions(${COMPONENT_LIB} PUBLIC KVM_GUD_PANEL_CHECK=1)
        endif()
    endif()
    if(KVM_USB_MOTION_CHECK)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE KVM_USB_MOTION_CHECK=1)
        if(CONFIG_IDF_TARGET_ESP32S3)
            target_sources(${COMPONENT_LIB} PRIVATE usb_compare_s3.S)
            target_compile_definitions(${COMPONENT_LIB} PRIVATE KVM_USB_SIMD_COMPARE=1)
        endif()
    endif()
endif()
if(KVM_USB_PICTURE_CHECK AND CONFIG_KVM_USB_MONITOR_EVAL)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE KVM_USB_PICTURE_CHECK=1)
endif()
